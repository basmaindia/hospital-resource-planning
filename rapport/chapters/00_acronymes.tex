\chapter*{Liste des Acronymes}
\addcontentsline{toc}{chapter}{Liste des Acronymes}

\begin{description}
    \item[API] Application Programming Interface
    \item[CBC] COIN-OR Branch and Cut (solveur libre d'optimisation en nombres entiers)
    \item[CHI] Service de Chirurgie
    \item[CNDP] Commission nationale de contrôle de la protection des données à caractère personnel
    \item[CORS] Cross-Origin Resource Sharing
    \item[CP-SAT] Constraint Programming -- Satisfiability (solveur Google OR-Tools)
    \item[DMS] Durée moyenne de séjour
    \item[ENSAM] École Nationale Supérieure d'Arts et Métiers
    \item[FHIR] Fast Healthcare Interoperability Resources (standard HL7)
    \item[HL7] Health Level Seven International
    \item[INDIA] Ingénierie Numérique en Data Science et Intelligence Artificielle
    \item[ISO] Organisation internationale de normalisation
    \item[KPI] Key Performance Indicator (indicateur clé de performance)
    \item[MAE] Mean Absolute Error (erreur absolue moyenne)
    \item[MED] Service de Médecine
    \item[MILP] Mixed-Integer Linear Programming (programmation linéaire en nombres entiers)
    \item[OBS] Organizational Breakdown Structure (structure organisationnelle)
    \item[RACI] Réalisateur, approbateur, consulté, informé
    \item[REA] Service de Réanimation
    \item[RH] Ressources humaines
    \item[RMSE] Root Mean Squared Error (racine de l'erreur quadratique moyenne)
    \item[SIH] Système d'information hospitalier
    \item[URG] Service des Urgences
    \item[WAPE] Weighted Absolute Percentage Error (erreur absolue pondérée en pourcentage)
    \item[WBS] Work Breakdown Structure (décomposition des travaux)
\end{description}
